    // CSV Parser
    function parseCSV(text) {
      const lines = [];
      let current = '';
      let inQuotes = false;
      const rows = [];

      for (let i = 0; i < text.length; i++) {
        const ch = text[i];
        if (inQuotes) {
          if (ch === '"' && text[i + 1] === '"') {
            current += '"';
            i++;
          } else if (ch === '"') {
            inQuotes = false;
          } else {
            current += ch;
          }
        } else {
          if (ch === '"') {
            inQuotes = true;
          } else if (ch === ',') {
            lines.push(current);
            current = '';
          } else if (ch === '\n' || ch === '\r') {
            lines.push(current);
            current = '';
            if (lines.length > 0 && lines.some(c => c.trim() !== '')) {
              rows.push([...lines]);
            }
            lines.length = 0;
            if (ch === '\r' && text[i + 1] === '\n') i++;
          } else {
            current += ch;
          }
        }
      }
      // Last line
      lines.push(current);
      if (lines.length > 0 && lines.some(c => c.trim() !== '')) {
        rows.push([...lines]);
      }

      return rows;
    }

    function handleDrop(e) {
      e.preventDefault();
      e.stopPropagation();
      document.getElementById('drop-zone').classList.remove('drop-zone-active');
      const file = e.dataTransfer.files[0];
      if (file) processFile(file);

    function handleFileSelect(e) {
      const file = e.target.files[0];
      if (file) processFile(file);
    }

    function processFile(file) {
      if (file.size > 50 * 1024 * 1024) {
        showToast('File exceeds 50MB limit', 'error');
        return;
      }
      if (!file.name.match(/\.(csv|txt)$/i)) {
        showToast('Please upload a .csv or .txt file', 'error');
        return;
      }

      currentFileName = file.name.replace(/\.[^.]+$/, '');

      const reader = new FileReader();
      reader.onload = function(e) {
        const text = e.target.result;
        const rows = parseCSV(text);

        if (rows.length < 2) {
          showToast('File appears empty or has no data rows', 'error');
          return;
        }

        csvHeaders = rows[0];
        parsedData = rows.slice(1);

        // Show file info
        document.getElementById('file-info').classList.remove('hidden');
        document.getElementById('file-name').textContent = file.name;
        document.getElementById('file-meta').textContent = `${parsedData.length} rows • ${csvHeaders.length} columns • ${formatFileSize(file.size)}`;

        // Stats
        document.getElementById('stats-section').classList.remove('hidden');
        document.getElementById('stat-rows').textContent = parsedData.length.toLocaleString();
        document.getElementById('stat-cols').textContent = csvHeaders.length;
        document.getElementById('stat-size').textContent = formatFileSize(file.size);

        // Preview table
        renderPreview();
        document.getElementById('preview-section').classList.remove('hidden');
        document.getElementById('row-count-badge').textContent = `Showing ${Math.min(parsedData.length, 50)} of ${parsedData.length} rows`;

        // Convert button
        document.getElementById('convert-section').classList.remove('hidden');

        showToast(`Loaded ${parsedData.length.toLocaleString()} rows successfully`, 'success');
      };
      reader.onerror = function() {
        showToast('Failed to read file', 'error');
      };
      reader.readAsText(file);
    }

        </tr>`
      ).join('');
    }

    function convertToXlsx() {
      const btn = document.getElementById('convert-btn');
      const btnText = document.getElementById('convert-btn-text');
      const origText = btnText.textContent;

      btn.disabled = true;
      btn.style.opacity = '0.7';
      btnText.innerHTML = '<span class="spinner" style="display:inline-block;"></span> Converting...';

      setTimeout(() => {
        try {
          const wsData = [csvHeaders, ...parsedData];
          const ws = XLSX.utils.aoa_to_sheet(wsData);

          // Auto-fit column widths
          const colWidths = csvHeaders.map((h, i) => {
            let maxLen = h.length;
            parsedData.slice(0, 100).forEach(row => {
              if (row[i] && row[i].length > maxLen) maxLen = row[i].length;
            });
            return { wch: Math.min(Math.max(maxLen + 2, 8), 40) };
          });
          ws['!cols'] = colWidths;

          const wb = XLSX.utils.book_new();
          XLSX.utils.book_append_sheet(wb, ws, 'Sheet1');
          XLSX.writeFile(wb, `${currentFileName}.xlsx`);

          showToast('XLSX file downloaded successfully!', 'success');
        } catch (err) {
          showToast('Conversion failed. Please try again.', 'error');
        } finally {
          btn.disabled = false;
          btn.style.opacity = '1';
          btnText.textContent = origText;
        }
      }, 600);
    }

    function formatFileSize(bytes) {
      if (bytes < 1024) return bytes + ' B';
      if (bytes < 1024 * 1024) return (bytes / 1024).toFixed(1) + ' KB';
      return (bytes / (1024 * 1024)).toFixed(1) + ' MB';
    }

    function escapeHtml(str) {
      const div = document.createElement('div');
      div.textContent = str || '';
      return div.innerHTML;
    }
